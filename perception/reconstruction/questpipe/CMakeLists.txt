# qrecon_quest: the reconstruction stages after tracking, for the headset (see tools/build_tracking.py).
cmake_minimum_required(VERSION 3.16)
project(qrecon_quest CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Eigen3 3.3 REQUIRED NO_MODULE)
find_package(OpenCV REQUIRED COMPONENTS core imgproc imgcodecs features2d calib3d)
if(OpenCV_VERSION VERSION_LESS 4.7)   # desktop test builds: ChArUco is in the contrib aruco module before 4.7
  find_package(OpenCV REQUIRED COMPONENTS core imgproc imgcodecs features2d calib3d aruco)
else()                                 # the headset's OpenCV 4.10: ChArUco is in objdetect
  find_package(OpenCV REQUIRED COMPONENTS core imgproc imgcodecs features2d calib3d objdetect)
endif()
set(NLOHMANN_JSON_INCLUDE "" CACHE PATH "folder holding nlohmann/json.hpp")
set(ORT_ROOT "" CACHE PATH "ONNX Runtime: folder with include/ (or headers/) and lib/")
find_path(ORT_INCLUDE onnxruntime_cxx_api.h PATHS ${ORT_ROOT}/include ${ORT_ROOT}/headers NO_DEFAULT_PATH NO_CMAKE_FIND_ROOT_PATH)
find_library(ORT_LIB onnxruntime PATHS ${ORT_ROOT}/lib ${ORT_ROOT}/jni/arm64-v8a NO_DEFAULT_PATH NO_CMAKE_FIND_ROOT_PATH)
add_executable(qrecon_quest qrecon_quest.cc)
target_include_directories(qrecon_quest PRIVATE ${NLOHMANN_JSON_INCLUDE})
target_link_libraries(qrecon_quest PRIVATE Eigen3::Eigen ${OpenCV_LIBS})
if(ORT_INCLUDE AND ORT_LIB)
  target_compile_definitions(qrecon_quest PRIVATE QRECON_WITH_ORT)
  target_include_directories(qrecon_quest PRIVATE ${ORT_INCLUDE})
  target_link_libraries(qrecon_quest PRIVATE ${ORT_LIB})
  message(STATUS "qrecon_quest: ONNX Runtime at ${ORT_LIB}")
else()
  message(FATAL_ERROR "ONNX Runtime not found under ORT_ROOT=${ORT_ROOT}")
endif()
install(TARGETS qrecon_quest RUNTIME DESTINATION bin)
