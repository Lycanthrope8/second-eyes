## Part 2 (RTX PC): the training suite, then both analyses from saved scores

```powershell
cd C:\Users\jhossai3\second-eyes
. {
    git pull origin master
    if ($LASTEXITCODE -ne 0) { throw "STOP: git pull" }
    if (-not (Select-String -Path "notes\decisions.md" -Pattern "^\| D114 \|" -Quiet)) { throw "STOP: this checkout lacks D114 (run Part 1 first)" }
    $DATA = "C:\Users\jhossai3\second-eyes-data"; $A24 = (Get-Content "$DATA\a24-current.txt").Trim(); $C = "$DATA\a26c-20261011-013341"
    $P26 = "$DATA\iref-release\prep-20261010-211017"
    $TR = ".\.venv-train\Scripts\python.exe"; $EV = ".\.venv\Scripts\python.exe"
    $T = "$DATA\tmp"; New-Item -ItemType Directory -Force $T | Out-Null; $env:TEMP = $T; $env:TMP = $T
    $env:PYTORCH_CUDA_ALLOC_CONF = "garbage_collection_threshold:0.8,max_split_size_mb:512"
    $V = "$DATA\a24-closure-tests-rtx.txt"
    & $TR grounding\tests\test_a24_train.py --samples "$DATA\iref-release-samples" --official "$DATA\iref-release\official" *>&1 | Set-Content $V
    if (Select-String -Path $V -Pattern "^\s*FAIL\s|Traceback|passed, [1-9]\d* failed" -CaseSensitive -Quiet) { throw "STOP: a test failed (see $V)" }
    Get-Content $V -Tail 1
    foreach ($f in "$A24\analysis-v2", "$C\analysis-v3") { if (Test-Path $f) { throw "STOP: $f already exists" } }
    $rec = (Get-Content "$C\analysis-v2\analysis.json" -Raw | ConvertFrom-Json).inputs.scores_manifest_sha256
    if ((Get-FileHash "$C\scores-v2\manifest.json" -Algorithm SHA256).Hash.ToLower() -ne $rec) { throw "STOP: A2.6c's scores-v2 is not the input its analysis-v2 recorded" }
    & $EV -m grounding.a24 analyze --prep $P26 --scores "$A24\scores" --out "$A24\analysis-v2" --supersedes "$A24\analysis"
    if ($LASTEXITCODE -ne 0) { throw "STOP: A2.4's analysis" }
    & $EV -m grounding.inference.iref_vla_dev analyze --prep $P26 --scores "$C\scores-v2" --desktop-compare "$C\desktop-compare" --out "$C\analysis-v3" --supersedes "$C\analysis-v2"
    if ($LASTEXITCODE -ne 0) { throw "STOP: A2.6c's analysis" }
    $diff = & $EV -m grounding.a24 analysis-diff --old "$C\analysis-v2" --new "$C\analysis-v3"
    if ($LASTEXITCODE -ne 0) { throw "STOP: the A2.6c comparison" }
    $diff
    $R = "$A24-closure"; if (Test-Path $R) { throw "STOP: $R already exists" }
    New-Item -ItemType Directory "$R\a24" | Out-Null; New-Item -ItemType Directory "$R\a26c" | Out-Null
    Copy-Item -Recurse "$A24\analysis-v2" "$R\a24"; Copy-Item -Recurse "$C\analysis-v3" "$R\a26c"; Copy-Item $V $R
    $diff | Set-Content -Encoding utf8 "$R\a26c\analysis-v3-diff.json"
    & $EV -c "import shutil, sys; print(shutil.make_archive(sys.argv[1], 'zip', sys.argv[1]))" $R
    if ($LASTEXITCODE -ne 0) { throw "STOP: the archive" }
    (Get-FileHash "$R.zip" -Algorithm SHA256).Hash.ToLower()
    "{0:N1} MB" -f ((Get-Item "$R.zip").Length / 1MB)
}
```
